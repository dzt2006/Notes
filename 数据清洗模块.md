```python
def print_format_stats(series, use_nfkc=True):
	data = series.copy()
    
    if use_nfkc：
    	data_str = data.apply(nfkc_strip)
        
    data_str = data.astype(str)
    
    def to_shape(match):
        return "#" * len(match.group())
    
    format_series = data_str.str.replace(r"\d+", to_shape, regex=True)
    
    format_stats = format_series.value_counts(dropna=False)
    
    print(format_stats.to_string())
```

```python
def date_clean(s):
    if pd.isna(s) or str(s).strip() == "":
        return np.nan
    s = str(s).strip()

    fmt_list = [
        "%Y-%m-%d",
        "%Y/%m/%d",
        "%Y_%m_%d",
    ]

    for fmt in fmt_list:
        try:
            dt = Datetime.strptime(s, fmt)
            return dt.strftime(""%Y-%m-%d"")
        except:
            continue

    return np.nan
```
