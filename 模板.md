# 爬虫通用模板

写爬虫时，可以先把下面这套模板固定下来：

```python
import csv
import time
import requests
from lxml import etree

session = requests.Session()
domain = "http://localhost:8000/"		# 域名或者地址

headers = {
    "User-Agent": "",
}

# 简单的去首位空格的函数
def clean(texts):
    return [t.strip() for t in texts if t and t.strip()]

columns = ["书名", "作者"]			    # 列头
csv_path = "books_all.csv"			   # 保存文件的名字


def crawl_all():
    current_url = domain + "page1.html"			# 当前页的地址
    all_books = []
    
    while True:
        resp = session.get(current_url, headers=headers, timeout=5)
        resp.encoding = "utf-8"
        html_dom = etree.HTML(resp.text)
        book_items = html_dom.xpath('//div[@class="book-item"]')
        page_books = []

        for item in book_items:
            book = {}

            # 书名
            title_list = clean(item.xpath('./h3[@class="title"]/text()'))
            book["书名"] = title_list[0] if title_list else ""

            # 作者
            author_list = clean(item.xpath('./p[@class="author"]/text()'))
            book["作者"] = author_list[0] if author_list else ""
        
        page_books.append(book)
        
        all_books.extend(page_books)
        
        print(f"已抓取：{current_url} | 本页{len(page_books)}本 | 累计{len(all_books)}本")

        
        #	找下一页的的按钮或者链接，直至找不到，断掉循环（此处代码省略）
		
        time.sleep(0.1)
    return all_books

def save_csv(book_list, path):
    with open(path, "w", encoding="utf-8-sig", newline="") as f:
        write = csv.DictWriter(f,fieldnames=columns)
        write.writeheader()
        write.writerows(book_list)
    print(f"CSV文件已保存：{path}")
    

if __name__ == "__main__":
    books = crawl_all()
    save_csv(books, csv_path)
```




下面补充一个取文本的细节：

```html

    <h3>Python入门</h3>

    </span>
</div>

```




`/text()`：只提取当前标签的直接子文本节点。

```python
res2 = html_dom.xpath('//span[@class="price"]//text()')
#	['\n        价格：\n        ', '￥', '149', '\n    ']
```




`//text()`：递归提取当前标签及所有后代标签里的全部文本节点。

```python
res3 = html_dom.xpath('string(//span[@class="price"])')
#	'\n        价格：\n        ￥\n        149\n    '
```




`string()`：提取当前标签及所有后代标签里的全部文本，并自动拼成一个完整字符串。

后两种情况不要直接对结果用 `clean(texts)`。如果一定要用，按下面的方式处理。

碰到 `//text()` 时：`clean()` 之后还要把结果拼接起来。

```python
clean_list = clean(res2)  # 结果: ['价格：', '￥', '149']

price_str = "".join(clean_list)  # 结果: '价格：￥149'

price = float(price_str.split("￥")[-1])
```




碰到 `string()` 时：它返回的是字符串而不是列表，要先用 `[]` 包成列表再传给 `clean()`。

```python
clean_list = clean([res3])

price = float(clean_list[0].split("￥")[-1])
```

统一文本格式，清理特殊空格和首尾空格。

```python
SPACE_PATTERN = r"[\u200b\u200c\u200d\u00a0\u3000\ufeff]"

def nfkc_strip(s):
    if pd.isna(s):
        return s
    s = str(s)
    s = unicodedata.normalize("NFKC", s)
    s = re.sub(SPACE_PATTERN, "", s)
    return s.strip()
```




统计一列数据的格式分布，快速发现异常格式。

```python
def print_format_stats(series,col_name="列",use_nfkc=True):
    s = series.copy()
    if use_nfkc:
        s = s.apply(nfkc_strip)

    fmt_series = s.map(lambda v: re.sub(r"\d+",lambda m: "#" * len(m.group()), str(v)))

    stat = fmt_series.value_counts(dropna=False).sort_values(ascending=False)

    print(f"{col_name} 格式分布：")
    print(stat.to_string())
```




对比清洗前后的缺失情况，检查是否有数据清洗失败。

```python
def clean_check(series,col_name, clean_func):
    raw = series.copy()
    cleaned = raw.apply(clean_func)

    extra_fail = (raw.notna() & raw.astype(str).str.strip().ne("") & cleaned.isna())

    print(
        f"{col_name}: 原始缺失={raw.isna().sum()}，"
        f"清洗后缺失={cleaned.isna().sum()}，"
        f"额外失败={extra_fail.sum()}"
    )

    return cleaned
```




## 分批写入数据库

```python
import pymysql

db_config = {
    "host": "localhost",
    "port": 3306,
    "user": "root",
    "password": "123456",
    "database": "spider_db",
    "charset": "utf8mb4"
}

batch_size = 500

insert_sql = """
			insert into books(书名, 作者)
			values(%s, %s)
			"""

# 在进行入库的时候，字段一定要清理成相应的字段
def books_rows(books):
    return [
        (
			b["书名"],b["作者"]        
        )
        for b in books
    ]
    
def save_mysql(books):
    conn = None
    try:
        conn = pymysql.connect(**db_config)
        rows = books_rows(books)
    	
        conn.begin()
        
        with conn.cursor() as cursor:
            cursor.execute("delete from books")
            for i in range(0, len(rows), batch_size):
                cursor.executemany(insert_sql, rows[i:i + batch_size])
                
        conn.commit()
        
        with conn.cursor() as cursor:
            cursor.execute("select count(*) from books")
            count = cursor.fetchone()[0]
            return count
     
    
    except Exception as e:
        if conn:
            conn.rollback()
        print(f"入库失败，已回滚：{e}")
        return None
    
    finally:
        if conn:
            conn.close()
            print("MySQL 连接已关闭")          
```
